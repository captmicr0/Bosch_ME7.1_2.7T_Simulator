cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(me71_bench_sim C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(me71_bench_sim main.c)
pico_generate_pio_header(me71_bench_sim ${CMAKE_CURRENT_LIST_DIR}/capture.pio)
pico_generate_pio_header(me71_bench_sim ${CMAKE_CURRENT_LIST_DIR}/dac_stream.pio)
pico_generate_pio_header(me71_bench_sim ${CMAKE_CURRENT_LIST_DIR}/cam_stream.pio)
target_link_libraries(me71_bench_sim pico_stdlib hardware_pio hardware_dma hardware_clocks hardware_spi pico_multicore)
pico_enable_stdio_usb(me71_bench_sim 1)
pico_enable_stdio_uart(me71_bench_sim 0)
pico_add_extra_outputs(me71_bench_sim)
